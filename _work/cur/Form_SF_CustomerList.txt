Option Compare Database
Option Explicit

Public Function OpenSelectedRecord() As Boolean

    On Error GoTo Err_Handler

    If Nz(Me!CustomerID, 0) <= 0 Then Exit Function

    DoCmd.OpenForm "F_CustomerSetup", acNormal, , "CustomerID = " & CLng(Me!CustomerID)

    OpenSelectedRecord = True

Exit_Handler:
    Exit Function

Err_Handler:
    MsgBox "The selected customer could not be opened." & vbCrLf & vbCrLf & _
           "Error Number: " & Err.Number & vbCrLf & _
           "Details: " & Err.Description, vbExclamation, "Open Customer"
    Resume Exit_Handler

End Function
