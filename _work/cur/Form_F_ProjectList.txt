Option Compare Database
Option Explicit

Private Const SUBFORM_CONTROL As String = "subProjectList"

Private Sub Form_Load()
    ApplyFilter
End Sub

Private Sub txtSearch_AfterUpdate()
    ApplyFilter
End Sub

Private Sub txtSearch_Change()
    ApplyFilter
End Sub

Private Sub txtSearch_KeyDown(KeyCode As Integer, Shift As Integer)

    If KeyCode = vbKeyReturn Then
        KeyCode = 0
        ApplyFilter
    End If

End Sub

Private Sub btnClearFilter_Click()

    Me.txtSearch = Null

    With Me.Controls(SUBFORM_CONTROL).Form
        .FilterOn = False
        .Filter = vbNullString
        .Requery
    End With

End Sub

Private Sub btnRefresh_Click()

    Me.Controls(SUBFORM_CONTROL).Form.Requery

End Sub

Private Sub btnNew_Click()

    OpenProjectFormNew

End Sub

Private Sub btnClose_Click()

    DoCmd.close acForm, Me.Name

End Sub

Private Sub ApplyFilter()

    On Error GoTo Err_Handler

    Dim searchText As String
    Dim filterText As String

    searchText = Trim$(Nz(Me.txtSearch, vbNullString))

    If Len(searchText) > 0 Then

        searchText = Replace(searchText, "'", "''")

        filterText = _
            "([ProjectCode] Like '*" & searchText & "*'" & _
            " OR [ProjectName] Like '*" & searchText & "*'" & _
            " OR [Status] Like '*" & searchText & "*'" & _
            " OR [CustomerPOReference] Like '*" & searchText & "*'" & _
            " OR [FinishedItemSKU] Like '*" & searchText & "*'" & _
            " OR [FinishedItemName] Like '*" & searchText & "*'" & _
            " OR [TemplateCode] Like '*" & searchText & "*'" & _
            " OR [TemplateName] Like '*" & searchText & "*'" & _
            " OR [OutputLocationCode] Like '*" & searchText & "*'" & _
            " OR [OutputLocationName] Like '*" & searchText & "*'" & _
            " OR [WarehouseName] Like '*" & searchText & "*')"

        With Me.Controls(SUBFORM_CONTROL).Form
            .Filter = filterText
            .FilterOn = True
            .Requery
        End With

    Else

        With Me.Controls(SUBFORM_CONTROL).Form
            .FilterOn = False
            .Filter = vbNullString
            .Requery
        End With

    End If

Exit_Handler:
    Exit Sub

Err_Handler:
    MsgBox _
        "The project list filter could not be applied." & vbCrLf & vbCrLf & _
        "Error Number: " & Err.Number & vbCrLf & _
        "Details: " & Err.Description, _
        vbExclamation, _
        "Project List Filter"

    Resume Exit_Handler

End Sub

Private Sub OpenProjectFormNew()

    On Error GoTo Err_Handler

    If FormExists("F_Projects") Then
        DoCmd.OpenForm "F_Projects", acNormal, , , acFormAdd
    ElseIf FormExists("F_Projects") Then
        DoCmd.OpenForm "F_Projects", acNormal, , , acFormAdd
    Else
        MsgBox _
            "No project detail form was found." & vbCrLf & vbCrLf & _
            "Expected one of these forms:" & vbCrLf & _
            "F_ProjectSetup" & vbCrLf & _
            "F_Project", _
            vbExclamation, _
            "Project Form Missing"
    End If

Exit_Handler:
    Exit Sub

Err_Handler:
    MsgBox _
        "The new project form could not be opened." & vbCrLf & vbCrLf & _
        "Error Number: " & Err.Number & vbCrLf & _
        "Details: " & Err.Description, _
        vbExclamation, _
        "New Project"

    Resume Exit_Handler

End Sub

Private Function FormExists(ByVal formName As String) As Boolean

    On Error GoTo Err_Handler

    Dim tmp As String
    tmp = CurrentProject.AllForms(formName).Name

    FormExists = True

Exit_Handler:
    Exit Function

Err_Handler:
    FormExists = False
    Resume Exit_Handler

End Function
