Private Sub Form_BeforeInsert(Cancel As Integer)

    On Error GoTo Err_Handler

    If Len(Trim$(Nz(Me.txtPONumber, vbNullString))) = 0 Then
        Me.txtPONumber = NextSupplierPONumber()
    End If

    If IsNull(Me.txtPODate) Then
        Me.txtPODate = Date
    End If

    If Len(Trim$(Nz(Me.cboPOType, vbNullString))) = 0 Then
        Me.cboPOType = "Original"
    End If

    If Len(Trim$(Nz(Me.cboPOStatus, vbNullString))) = 0 Then
        Me.cboPOStatus = "Draft"
    End If

    If IsNull(Me.chkPrepaidFreight) Then
        Me.chkPrepaidFreight = False
    End If

    If IsNull(Me.txtSubtotalAmount) Then
        Me.txtSubtotalAmount = 0
    End If

    If IsNull(Me.txtFreightAmount) Then
        Me.txtFreightAmount = 0
    End If

    If IsNull(Me.txtTaxAmount) Then
        Me.txtTaxAmount = 0
    End If

    If IsNull(Me.txtTotalAmount) Then
        Me.txtTotalAmount = 0
    End If

    If IsNull(Me.cboCurrencyID) Then
        If DCount("*", "T_Currencies") = 1 Then
            Me.cboCurrencyID = DLookup("CurrencyID", "T_Currencies")
        End If
    End If

    If IsNull(Me.txtCreatedOn) Then
        Me.txtCreatedOn = Now()
    End If

Exit_Handler:
    Exit Sub

Err_Handler:
    Cancel = True
    MsgBox _
        "The new Supplier PO could not be started." & _
        vbCrLf & vbCrLf & _
        "Error Number: " & Err.Number & vbCrLf & _
        "Details: " & Err.Description, _
        vbExclamation, _
        "New Supplier PO"
    Resume Exit_Handler

End Sub


Private Sub Form_BeforeUpdate(Cancel As Integer)
    UX_CoerceBooleans Me

    On Error GoTo Err_Handler

    NormalizeHeaderValues

    If Not ValidateBasicHeader Then
        Cancel = True
        Exit Sub
    End If

    If IsNull(Me.txtCreatedOn) Then
        Me.txtCreatedOn = Now()
    End If

    Me.txtUpdatedOn = Now()

Exit_Handler:
    Exit Sub

Err_Handler:
    Cancel = True
    MsgBox _
        "The Supplier PO header could not be validated." & _
        vbCrLf & vbCrLf & _
        "Error Number: " & Err.Number & vbCrLf & _
        "Details: " & Err.Description, _
        vbExclamation, _
        "Supplier PO Validation"
    Resume Exit_Handler

End Sub


Private Sub Form_AfterUpdate()

    On Error Resume Next

    ConfigureLineSubform
    Me.Controls(LINE_SUBFORM).Form.Requery
    RefreshWorkflowState

    On Error GoTo 0

End Sub


Private Sub Form_Error(DataErr As Integer, Response As Integer)
    Dim uxHandled As Boolean
    UX_SuppressFormError DataErr, Response, uxHandled
    If uxHandled Then Exit Sub

    Select Case DataErr

        Case 3022
            MsgBox _
                "That Supplier PO number already exists." & _
                vbCrLf & _
                "A new PO number will be generated when you try again.", _
                vbExclamation, _
                "Duplicate Supplier PO"

            If Me.NewRecord Then
                On Error Resume Next
                Me.txtPONumber = NextSupplierPONumber()
                On Error GoTo 0
            End If

            Response = acDataErrContinue

        Case 3314
            MsgBox _
                "A required Supplier PO field is missing.", _
                vbExclamation, _
                "Incomplete Supplier PO"

            Response = acDataErrContinue

        Case Else
            Response = acDataErrDisplay

    End Select

End Sub

Private Sub cboSupplierID_BeforeUpdate(Cancel As Integer)

    On Error GoTo Exit_Handler

    If Nz(Me.txtSupplierPOID, 0) <= 0 Then
        Exit Sub
    End If

    If Nz(Me.cboSupplierID.OldValue, 0) <= 0 Then
        Exit Sub
    End If

    If Nz(Me.cboSupplierID.OldValue, 0) = _
       Nz(Me.cboSupplierID.value, 0) Then
        Exit Sub
    End If

    If DCount( _
        "*", _
        "T_SupplierPOLines", _
        "SupplierPOID=" & CLng(Me.txtSupplierPOID)) > 0 Then

        If MsgBox( _
            "This PO already contains item lines." & _
            vbCrLf & vbCrLf & _
            "Changing the supplier can make existing supplier part " & _
            "numbers and suggested prices incorrect." & _
            vbCrLf & vbCrLf & _
            "Continue changing the supplier?", _
            vbQuestion + vbYesNo + vbDefaultButton2, _
            "Change Supplier") <> vbYes Then

            Cancel = True
        End If
    End If

Exit_Handler:
    Exit Sub

End Sub

'=============================================================
' PUBLIC TOTAL RECALCULATION USED BY THE LINE SUBFORM
'=============================================================
Public Sub RecalculatePOTotals()

    On Error GoTo Err_Handler

    Dim SupplierPOID As Long
    Dim calculatedSubtotal As Currency
    Dim calculatedTax As Currency
    Dim FreightAmount As Currency
    Dim calculatedTotal As Currency

    SupplierPOID = CurrentSupplierPOID()

    If SupplierPOID > 0 Then
        calculatedSubtotal = Nz(DLookup( _
            "CalculatedSubtotal", _
            "Q_SupplierPOLineTotals", _
            "SupplierPOID=" & SupplierPOID), 0)

        calculatedTax = Nz(DLookup( _
            "CalculatedTax", _
            "Q_SupplierPOLineTotals", _
            "SupplierPOID=" & SupplierPOID), 0)
    End If

    FreightAmount = Nz(Me.txtFreightAmount, 0)
    calculatedTotal = _
        calculatedSubtotal + calculatedTax + FreightAmount

    SetCurrencyIfDifferent _
        Me.txtSubtotalAmount, _
        calculatedSubtotal

    SetCurrencyIfDifferent _
        Me.txtTaxAmount, _
        calculatedTax

    SetCurrencyIfDifferent _
        Me.txtTotalAmount, _
        calculatedTotal

    Me.Recalc

Exit_Handler:
    Exit Sub

Err_Handler:
    MsgBox _
        "The Supplier PO totals could not be recalculated." & _
        vbCrLf & vbCrLf & _
        "Error Number: " & Err.Number & vbCrLf & _
        "Details: " & Err.Description, _
        vbExclamation, _
        "Supplier PO Totals"
    Resume Exit_Handler

End Sub

'=============================================================
' VALIDATION
'=============================================================
Private Function ValidateBasicHeader() As Boolean

    ValidateBasicHeader = False

    If Len(Trim$(Nz(Me.txtPONumber, vbNullString))) = 0 Then
        Me.txtPONumber = NextSupplierPONumber()
    End If

    If DCount( _
        "*", _
        "T_SupplierPOs", _
        "PONumber='" & _
        Replace(Me.txtPONumber, "'", "''") & _
        "' AND SupplierPOID<>" & _
        Nz(Me.txtSupplierPOID, 0)) > 0 Then

        MsgBox _
            "Supplier PO number '" & Me.txtPONumber & _
            "' already exists.", _
            vbExclamation, _
            "Duplicate Supplier PO"
        Exit Function
    End If

    If Nz(Me.cboSupplierID, 0) <= 0 Then
        MsgBox _
            "Select the supplier for this purchase order.", _
            vbExclamation, _
            "Supplier Required"
        Me.cboSupplierID.SetFocus
        Exit Function
    End If

    If IsNull(Me.txtPODate) Then
        MsgBox _
            "Enter the PO date.", _
            vbExclamation, _
            "PO Date Required"
        Me.txtPODate.SetFocus
        Exit Function
    End If

    If Not IsNull(Me.txtExpectedDeliveryDate) Then
        If CDate(Me.txtExpectedDeliveryDate) < _
           CDate(Me.txtPODate) Then

            MsgBox _
                "Expected Delivery Date cannot be earlier than " & _
                "the PO Date.", _
                vbExclamation, _
                "Invalid Delivery Date"
            Me.txtExpectedDeliveryDate.SetFocus
            Exit Function
        End If
    End If

    If Nz(Me.txtFreightAmount, 0) < 0 Then
        MsgBox _
            "Freight Amount cannot be negative.", _
            vbExclamation, _
            "Invalid Freight Amount"
        Me.txtFreightAmount.SetFocus
        Exit Function
    End If

    ValidateBasicHeader = True

End Function


Private Function ValidatePOForSending() As Boolean

    Dim SupplierPOID As Long

    ValidatePOForSending = False
    SupplierPOID = CurrentSupplierPOID()

    If SupplierPOID <= 0 Then
        Exit Function
    End If

    If DCount( _
        "*", _
        "T_SupplierPOLines", _
        "SupplierPOID=" & SupplierPOID & _
        " AND Nz([LineStatus],'Open')<>'Cancelled'") = 0 Then

        MsgBox _
            "Add at least one active item line before sending the PO.", _
            vbExclamation, _
            "PO Lines Required"
        Exit Function
    End If

    If Nz(Me.cboDeliveryLocationID, 0) <= 0 Then
        MsgBox _
            "Select the delivery location before sending the PO.", _
            vbExclamation, _
            "Delivery Location Required"
        Me.cboDeliveryLocationID.SetFocus
        Exit Function
    End If

    If Nz(Me.cboCurrencyID, 0) <= 0 Then
        MsgBox _
            "Select the PO currency before sending it.", _
            vbExclamation, _
            "Currency Required"
        Me.cboCurrencyID.SetFocus
        Exit Function
    End If

    If Len(Trim$(Nz(Me.txtShipToName, vbNullString))) = 0 Then
        MsgBox _
            "Enter the Ship To name before sending the PO.", _
            vbExclamation, _
            "Ship To Required"
        Me.txtShipToName.SetFocus
        Exit Function
    End If

    If Len(Trim$(Nz(Me.txtShipToAddress, vbNullString))) = 0 Then
        MsgBox _
            "Enter the Ship To address before sending the PO.", _
            vbExclamation, _
            "Ship To Address Required"
        Me.txtShipToAddress.SetFocus
        Exit Function
    End If

    If Len(Trim$(Nz(Me.txtAuthorizedBy, vbNullString))) = 0 Then
        MsgBox _
            "Enter the person authorizing this PO.", _
            vbExclamation, _
            "Authorization Required"
        Me.txtAuthorizedBy.SetFocus
        Exit Function
    End If

    ValidatePOForSending = True

End Function

'=============================================================
' PO NUMBER GENERATION
'=============================================================
Private Function NextSupplierPONumber() As String

    On Error GoTo Err_Handler

    Dim rs As DAO.Recordset
    Dim existingNumber As String
    Dim numericPart As Long
    Dim maximumNumber As Long
    Dim candidate As String

    Set rs = CurrentDb.OpenRecordset( _
        "SELECT PONumber " & _
        "FROM T_SupplierPOs " & _
        "WHERE PONumber Like 'SPO-*';", _
        dbOpenSnapshot)

    Do While Not rs.EOF

        existingNumber = Trim$(Nz(rs!PONumber, vbNullString))

        If UCase$(Left$(existingNumber, 4)) = "SPO-" Then
            numericPart = Val(Mid$(existingNumber, 5))

            If numericPart > maximumNumber Then
                maximumNumber = numericPart
            End If
        End If

        rs.MoveNext
    Loop

    candidate = "SPO-" & Format$(maximumNumber + 1, "00000")

    Do While DCount( _
        "*", _
        "T_SupplierPOs", _
        "PONumber='" & Replace(candidate, "'", "''") & "'") > 0

        maximumNumber = maximumNumber + 1
        candidate = "SPO-" & Format$(maximumNumber + 1, "00000")
    Loop

    NextSupplierPONumber = candidate

Exit_Handler:
    On Error Resume Next
    If Not rs Is Nothing Then
        rs.close
    End If
    Set rs = Nothing
    On Error GoTo 0
    Exit Function

Err_Handler:
    NextSupplierPONumber = _
        "SPO-" & Format$(Nz(DMax( _
            "SupplierPOID", _
            "T_SupplierPOs"), 0) + 1, "00000")
    Resume Exit_Handler

End Function

Private Sub SetCurrencyIfDifferent( _
    ByVal targetControl As Access.Control, _
    ByVal newValue As Currency _
)

    If CCur(Nz(targetControl.value, 0)) <> newValue Then
        targetControl.value = newValue
    End If

End Sub


Private Sub NormalizeTextControl(ByVal controlName As String)

    If Not ControlExists(controlName) Then
        Exit Sub
    End If

    With Me.Controls(controlName)
        If Len(Trim$(Nz(.value, vbNullString))) = 0 Then
            .value = Null
        Else
            .value = Trim$(CStr(.value))
        End If
    End With

End Sub


Private Function ControlExists(ByVal controlName As String) As Boolean

    On Error GoTo Not_Found

    Dim controlObject As Access.Control

    Set controlObject = Me.Controls(controlName)
    ControlExists = True

Exit_Handler:
    Set controlObject = Nothing
    Exit Function

Not_Found:
    ControlExists = False
    Resume Exit_Handler

End Function


Private Function FormExists(ByVal formName As String) As Boolean

    On Error GoTo Not_Found

    Dim formObject As AccessObject

    For Each formObject In CurrentProject.AllForms
        If StrComp( _
            formObject.Name, _
            formName, _
            vbTextCompare) = 0 Then

            FormExists = True
            Exit Function
        End If
    Next formObject

Not_Found:
    FormExists = False

End Function
