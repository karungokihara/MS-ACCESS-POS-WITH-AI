Private Sub Form_Load()
    UX_MarkRequired Me, "CustomerID", "InvoiceDate"

    On Error Resume Next

    ConfigureInvoiceForm
    ConfigureInvoiceCombos
    SetInvoiceDefaults
    SetLineSubformAvailability

    On Error GoTo 0

End Sub


Private Sub Form_Current()

    On Error Resume Next

    SetInvoiceDefaults
    SetLineSubformAvailability

    On Error GoTo 0

End Sub


Private Sub Form_BeforeUpdate(Cancel As Integer)
    UX_CoerceBooleans Me

    On Error GoTo Err_Handler

    If m_IsDiscardingEdits Then
        Exit Sub
    End If


    If IsBlankNewInvoiceRecord Then
        Cancel = True
        Exit Sub
    End If

    NormalizeInvoiceValues

    If Not ValidateInvoiceRecord Then
        Cancel = True
        Exit Sub
    End If

    If Me.NewRecord Then
        SafeSetControlValue "CreatedOn", Now()
    End If

    SafeSetControlValue "UpdatedOn", Now()

Exit_Handler:
    Exit Sub

Err_Handler:

    Cancel = True

    MsgBox _
        "The invoice could not be validated." & _
        vbCrLf & vbCrLf & _
        "Error " & Err.Number & ": " & Err.Description, _
        vbExclamation, _
        "Invoice Validation"

'=============================================================
' SAVE INVOICE
'=============================================================

Private Function SaveInvoiceRecord( _
    Optional ByVal showSuccessMessage As Boolean = True _
) As Boolean

    On Error GoTo Err_Handler

    SaveInvoiceRecord = False

    If IsBlankNewInvoiceRecord Then

        MsgBox _
            "Enter invoice details before saving.", _
            vbExclamation, _
            "Invoice Required"

        SafeSetFocus "CustomerID"
        Exit Function

    End If

    m_IsExplicitSave = True

    NormalizeInvoiceValues

    If Not ValidateInvoiceRecord Then
        m_IsExplicitSave = False
        Exit Function
    End If

    If Me.NewRecord Then
        SafeSetControlValue "CreatedOn", Now()
    End If

    SafeSetControlValue "UpdatedOn", Now()

    If Me.Dirty Then
        DoCmd.RunCommand acCmdSaveRecord
    End If

    If SaveInvoiceLineEditsSafely(False) = False Then
        m_IsExplicitSave = False
        Exit Function
    End If

    CalculateAndStoreInvoiceTotals False

    If Me.Dirty Then
        DoCmd.RunCommand acCmdSaveRecord
    End If

    m_IsExplicitSave = False

    If showSuccessMessage Then

        MsgBox _
            "Invoice saved successfully.", _
            vbInformation, _
            "Invoice Saved"

    End If

    SaveInvoiceRecord = True

Exit_Handler:
    m_IsExplicitSave = False
    Exit Function

Err_Handler:

    m_IsExplicitSave = False

    If Err.Number = 3200 Then

        MsgBox _
            "The invoice could not be saved because Access detected an unsafe parent invoice change while invoice lines exist." & _
            vbCrLf & vbCrLf & _
            "Cancel the pending header edit, then save again.", _
            vbExclamation, _
            "Invoice Relationship Protection"

    ElseIf Err.Number = 3022 Then

        MsgBox _
            "This invoice number already exists. Use a different invoice number.", _
            vbExclamation, _
            "Duplicate Invoice"

    Else

        MsgBox _
            "The invoice could not be saved." & _
            vbCrLf & vbCrLf & _
            "Error " & Err.Number & ": " & Err.Description, _
            vbExclamation, _
            "Invoice Save Failed"

    End If

    SaveInvoiceRecord = False
    Resume Exit_Handler

End Function

Private Sub ExportCurrentInvoiceReportToPDF()

    On Error GoTo Err_Handler

    Dim invoiceIDValue As Long
    Dim invoiceNoValue As String
    Dim folderPath As String
    Dim FilePath As String
    Dim whereText As String

    Dim realErrNumber As Long
    Dim realErrDescription As String

    invoiceIDValue = GetCurrentInvoiceID()
    invoiceNoValue = Trim$(Nz(GetControlValue("InvoiceNo"), vbNullString))

    If Len(invoiceNoValue) = 0 Then
        invoiceNoValue = "Invoice_" & CStr(invoiceIDValue)
    End If

    invoiceNoValue = CleanFileName(invoiceNoValue)

    folderPath = Environ$("USERPROFILE") & PDF_INVOICE_FOLDER

    EnsureFolderExists Environ$("USERPROFILE") & PDF_MAIN_FOLDER
    EnsureFolderExists folderPath

    FilePath = folderPath & "\Invoice_" & invoiceNoValue & ".pdf"

    whereText = "[InvoiceID]=" & invoiceIDValue

    If ReportExists(INVOICE_REPORT_NAME) = False Then

        MsgBox _
            "The invoice report was not found." & vbCrLf & vbCrLf & _
            "Expected report: " & INVOICE_REPORT_NAME, _
            vbExclamation, _
            "Invoice Report Missing"

        Exit Sub

    End If

    If CurrentProject.AllReports(INVOICE_REPORT_NAME).IsLoaded Then
        DoCmd.close acReport, INVOICE_REPORT_NAME, acSaveNo
    End If

    DoCmd.OpenReport _
        INVOICE_REPORT_NAME, _
        acViewPreview, _
        , _
        whereText, _
        acHidden

    DoCmd.OutputTo _
        acOutputReport, _
        INVOICE_REPORT_NAME, _
        acFormatPDF, _
        FilePath, _
        True

    DoCmd.close acReport, INVOICE_REPORT_NAME, acSaveNo

    MsgBox _
        "Invoice PDF created successfully." & _
        vbCrLf & vbCrLf & _
        FilePath, _
        vbInformation, _
        "Create Invoice PDF"

Exit_Handler:
    Exit Sub

Err_Handler:

    realErrNumber = Err.Number
    realErrDescription = Err.Description

    On Error Resume Next
    DoCmd.close acReport, INVOICE_REPORT_NAME, acSaveNo
    On Error GoTo 0

    MsgBox _
        "The invoice PDF could not be created." & _
        vbCrLf & vbCrLf & _
        "Error Number: " & realErrNumber & vbCrLf & _
        "Details: " & realErrDescription, _
        vbExclamation, _
        "Create Invoice PDF"

    Resume Exit_Handler

End Sub


'=============================================================
' LINK RESOLUTION
'=============================================================

Private Function ResolveSalesOrderID(ByVal invoiceIDValue As Long) As Long

    On Error GoTo Err_Handler

    Dim foundValue As Variant

    ResolveSalesOrderID = 0

    If invoiceIDValue <= 0 Then Exit Function

    '1. Directly from T_Invoices
    foundValue = SafeDLookup("SalesOrderID", "T_Invoices", "InvoiceID=" & invoiceIDValue)

    If Nz(foundValue, 0) > 0 Then
        ResolveSalesOrderID = CLng(foundValue)
        Exit Function
    End If

    '2. From current form control
    foundValue = GetControlValue("SalesOrderID")

    If Nz(foundValue, 0) > 0 Then
        ResolveSalesOrderID = CLng(foundValue)
        Exit Function
    End If

    '3. From open Sales Order form
    If FormIsLoaded("F_SalesOrders") Then

        foundValue = Nz(Forms!F_SalesOrders!SalesOrderID, 0)

        If Nz(foundValue, 0) > 0 Then
            ResolveSalesOrderID = CLng(foundValue)
            Exit Function
        End If

    End If

    '4. From invoice lines through SalesOrderLineID
    foundValue = SafeDLookup( _
        "SalesOrderLineID", _
        "T_InvoiceLines", _
        "InvoiceID=" & invoiceIDValue & " AND SalesOrderLineID Is Not Null" _
    )

    If Nz(foundValue, 0) > 0 Then

        foundValue = SafeDLookup( _
            "SalesOrderID", _
            "T_SalesOrderLines", _
            "SalesOrderLineID=" & CLng(foundValue) _
        )

        If Nz(foundValue, 0) > 0 Then
            ResolveSalesOrderID = CLng(foundValue)
            Exit Function
        End If

    End If

Exit_Handler:
    Exit Function

Err_Handler:
    ResolveSalesOrderID = 0
    Resume Exit_Handler

End Function


Private Function ResolveCustomerPOID( _
    ByVal invoiceIDValue As Long, _
    ByVal salesOrderIDValue As Long _
) As Long

    On Error GoTo Err_Handler

    Dim foundValue As Variant
    Dim salesOrderLineIDValue As Long
    Dim customerPOLineIDValue As Long

    ResolveCustomerPOID = 0

    If invoiceIDValue <= 0 Then Exit Function

    '1. Directly from T_Invoices
    foundValue = SafeDLookup("CustomerPOID", "T_Invoices", "InvoiceID=" & invoiceIDValue)

    If Nz(foundValue, 0) > 0 Then
        ResolveCustomerPOID = CLng(foundValue)
        Exit Function
    End If

    '2. From current form control
    foundValue = GetControlValue("CustomerPOID")

    If Nz(foundValue, 0) > 0 Then
        ResolveCustomerPOID = CLng(foundValue)
        Exit Function
    End If

    '3. From open Customer PO form
    If FormIsLoaded("F_CustomerPO") Then

        foundValue = Nz(Forms!F_CustomerPO!CustomerPOID, 0)

        If Nz(foundValue, 0) > 0 Then
            ResolveCustomerPOID = CLng(foundValue)
            Exit Function
        End If

    End If

    '4. From Sales Order header
    If salesOrderIDValue > 0 Then

        foundValue = SafeDLookup( _
            "CustomerPOID", _
            "T_SalesOrders", _
            "SalesOrderID=" & salesOrderIDValue _
        )

        If Nz(foundValue, 0) > 0 Then
            ResolveCustomerPOID = CLng(foundValue)
            Exit Function
        End If

    End If

    '5. From invoice line CustomerPOLineID
    foundValue = SafeDLookup( _
        "CustomerPOLineID", _
        "T_InvoiceLines", _
        "InvoiceID=" & invoiceIDValue & " AND CustomerPOLineID Is Not Null" _
    )

    If Nz(foundValue, 0) > 0 Then

        customerPOLineIDValue = CLng(foundValue)

        foundValue = SafeDLookup( _
            "CustomerPOID", _
            "T_CustomerPOLines", _
            "CustomerPOLineID=" & customerPOLineIDValue _
        )

        If Nz(foundValue, 0) > 0 Then
            ResolveCustomerPOID = CLng(foundValue)
            Exit Function
        End If

    End If

    '6. From invoice line SalesOrderLineID -> SalesOrderLine -> CustomerPOLineID -> CustomerPO
    foundValue = SafeDLookup( _
        "SalesOrderLineID", _
        "T_InvoiceLines", _
        "InvoiceID=" & invoiceIDValue & " AND SalesOrderLineID Is Not Null" _
    )

    If Nz(foundValue, 0) > 0 Then

        salesOrderLineIDValue = CLng(foundValue)

        foundValue = SafeDLookup( _
            "CustomerPOLineID", _
            "T_SalesOrderLines", _
            "SalesOrderLineID=" & salesOrderLineIDValue _
        )

        If Nz(foundValue, 0) > 0 Then

            customerPOLineIDValue = CLng(foundValue)

            foundValue = SafeDLookup( _
                "CustomerPOID", _
                "T_CustomerPOLines", _
                "CustomerPOLineID=" & customerPOLineIDValue _
            )

            If Nz(foundValue, 0) > 0 Then
                ResolveCustomerPOID = CLng(foundValue)
                Exit Function
            End If

        End If

    End If

Exit_Handler:
    Exit Function

Err_Handler:
    ResolveCustomerPOID = 0
    Resume Exit_Handler

End Function


Private Sub SyncInvoiceLinks( _
    ByVal invoiceIDValue As Long, _
    ByVal salesOrderIDValue As Long, _
    ByVal customerPOIDValue As Long _
)

    On Error Resume Next

    Dim SqlText As String
    Dim setText As String

    If invoiceIDValue <= 0 Then Exit Sub

    If salesOrderIDValue > 0 Then
        setText = setText & "SalesOrderID=" & salesOrderIDValue
    End If

    If customerPOIDValue > 0 Then

        If Len(setText) > 0 Then
            setText = setText & ", "
        End If

        setText = setText & "CustomerPOID=" & customerPOIDValue

    End If

    If Len(setText) = 0 Then Exit Sub

    setText = setText & ", UpdatedOn=#" & Format$(Now(), "yyyy\/mm\/dd hh\:nn\:ss") & "#"

    SqlText = _
        "UPDATE T_Invoices SET " & setText & _
        " WHERE InvoiceID=" & invoiceIDValue & ";"

    CurrentDb.Execute SqlText, 128

    If salesOrderIDValue > 0 Then
        SafeSetControlValue "SalesOrderID", salesOrderIDValue
    End If

    If customerPOIDValue > 0 Then
        SafeSetControlValue "CustomerPOID", customerPOIDValue
    End If

    On Error GoTo 0

End Sub

End Function

Private Function CalculateAndStoreInvoiceTotals( _
    Optional ByVal showErrors As Boolean = False _
) As Boolean

    On Error GoTo Err_Handler

    Dim invoiceIDValue As Long
    Dim subtotalValue As Currency
    Dim taxValue As Currency
    Dim grandTotalValue As Currency
    Dim sqlUpdate As String

    CalculateAndStoreInvoiceTotals = False

    invoiceIDValue = GetCurrentInvoiceID()

    If invoiceIDValue <= 0 Then
        CalculateAndStoreInvoiceTotals = True
        Exit Function
    End If

    subtotalValue = Nz(DSum( _
        "Nz([LineSubtotal],0)", _
        "T_InvoiceLines", _
        "InvoiceID = " & invoiceIDValue _
    ), 0)

    taxValue = Nz(GetControlValue("TaxAmount"), 0)
    grandTotalValue = subtotalValue + taxValue

    sqlUpdate = _
        "UPDATE T_Invoices SET " & _
        "SubtotalAmount = " & SqlCurrency(subtotalValue) & ", " & _
        "GrandTotalAmount = " & SqlCurrency(grandTotalValue) & ", " & _
        "UpdatedOn = #" & Format$(Now(), "yyyy\/mm\/dd hh\:nn\:ss") & "# " & _
        "WHERE InvoiceID = " & invoiceIDValue & ";"

    CurrentDb.Execute sqlUpdate, 128

    SafeSetControlValue "SubtotalAmount", subtotalValue
    SafeSetControlValue "GrandTotalAmount", grandTotalValue

    CalculateAndStoreInvoiceTotals = True

Exit_Handler:
    Exit Function

Err_Handler:

    CalculateAndStoreInvoiceTotals = False

    If showErrors Then
        MsgBox _
            "Invoice totals could not be recalculated." & _
            vbCrLf & vbCrLf & _
            "Error " & Err.Number & ": " & Err.Description, _
            vbExclamation, _
            "Invoice Totals"
    End If

    Resume Exit_Handler

End Function


Private Function SqlCurrency(ByVal valueToFormat As Currency) As String

    SqlCurrency = Trim$(Str$(CDbl(valueToFormat)))

End Function

Private Sub NormalizeInvoiceValues()

    If Len(Trim$(Nz(GetControlValue("InvoiceNo"), vbNullString))) = 0 Then
        SafeSetControlValue "InvoiceNo", GetNextInvoiceNumber()
    Else
        SafeSetControlValue "InvoiceNo", UCase$(Trim$(Nz(GetControlValue("InvoiceNo"), vbNullString)))
    End If

    If IsNull(GetControlValue("InvoiceDate")) Then
        SafeSetControlValue "InvoiceDate", Date
    End If

    If Len(Trim$(Nz(GetControlValue("InvoiceType"), vbNullString))) = 0 Then
        SafeSetControlValue "InvoiceType", "Pro Forma"
    End If

    If Len(Trim$(Nz(GetControlValue("InvoiceStatus"), vbNullString))) = 0 Then
        SafeSetControlValue "InvoiceStatus", "Draft"
    End If

    SafeSetControlValue "PaymentTerms", Trim$(Nz(GetControlValue("PaymentTerms"), vbNullString))
    SafeSetControlValue "BillToName", Trim$(Nz(GetControlValue("BillToName"), vbNullString))
    SafeSetControlValue "BillToAddress", Trim$(Nz(GetControlValue("BillToAddress"), vbNullString))
    SafeSetControlValue "Notes", Trim$(Nz(GetControlValue("Notes"), vbNullString))

    If IsNull(GetControlValue("TaxAmount")) Then
        SafeSetControlValue "TaxAmount", 0
    End If

End Sub


Private Function ValidateInvoiceRecord() As Boolean

    On Error GoTo Err_Handler

    Dim currentInvoiceID As Long
    Dim invoiceNoValue As String
    Dim criteriaText As String

    ValidateInvoiceRecord = False

    currentInvoiceID = GetCurrentInvoiceID()
    invoiceNoValue = Trim$(Nz(GetControlValue("InvoiceNo"), vbNullString))

    If Len(invoiceNoValue) = 0 Then

        MsgBox _
            "Enter the invoice number.", _
            vbExclamation, _
            "Invoice Number Required"

        SafeSetFocus "InvoiceNo"
        Exit Function

    End If

    If IsNull(GetControlValue("CustomerID")) Then

        MsgBox _
            "Select the customer.", _
            vbExclamation, _
            "Customer Required"

        SafeSetFocus "CustomerID"
        Exit Function

    End If

    If IsNull(GetControlValue("InvoiceDate")) Then

        MsgBox _
            "Enter the invoice date.", _
            vbExclamation, _
            "Invoice Date Required"

        SafeSetFocus "InvoiceDate"
        Exit Function

    End If

    criteriaText = "InvoiceNo = '" & Replace(invoiceNoValue, "'", "''") & "'"

    If currentInvoiceID > 0 Then
        criteriaText = criteriaText & " AND InvoiceID <> " & currentInvoiceID
    End If

    If DCount("*", "T_Invoices", criteriaText) > 0 Then

        MsgBox _
            "This invoice number already exists.", _
            vbExclamation, _
            "Duplicate Invoice"

        SafeSetFocus "InvoiceNo"
        Exit Function

    End If

    ValidateInvoiceRecord = True

Exit_Handler:
    Exit Function

Err_Handler:
    ValidateInvoiceRecord = False
    Resume Exit_Handler

End Function

Private Function SaveInvoiceLineEditsSafely( _
    Optional ByVal showErrors As Boolean = False _
) As Boolean

    On Error GoTo Err_Handler

    Dim frmLines As Form

    SaveInvoiceLineEditsSafely = True

    If Not ControlExists("subInvoiceLines") Then
        Exit Function
    End If

    Set frmLines = Me.Controls("subInvoiceLines").Form

    If frmLines.Dirty Then

        If SubformInvoiceLineIsBlank(frmLines) Then
            frmLines.Undo
        Else
            frmLines.Dirty = False
        End If

    End If

Exit_Handler:
    Set frmLines = Nothing
    Exit Function

Err_Handler:

    If IsAutomationDisconnectError(Err.Number) Then
        SaveInvoiceLineEditsSafely = True
        Resume Exit_Handler
    End If

    SaveInvoiceLineEditsSafely = False

    If showErrors Then
        MsgBox _
            "The invoice line currently being edited could not be saved." & _
            vbCrLf & vbCrLf & _
            "Error " & Err.Number & ": " & Err.Description, _
            vbExclamation, _
            "Invoice Line Save Failed"
    End If

    Resume Exit_Handler

End Function


Private Function SubformInvoiceLineIsBlank(ByVal frm As Form) As Boolean

    On Error GoTo Err_Handler

    If Not frm.NewRecord Then
        SubformInvoiceLineIsBlank = False
        Exit Function
    End If

    SubformInvoiceLineIsBlank = _
        IsNull(frm!ItemID) And _
        Len(Trim$(Nz(frm!ItemCode, vbNullString))) = 0 And _
        Len(Trim$(Nz(frm!LineDescription, vbNullString))) = 0 And _
        Nz(frm!Quantity, 0) = 0 And _
        IsNull(frm!UnitID) And _
        Nz(frm!UnitPrice, 0) = 0

    Exit Function

Err_Handler:
    SubformInvoiceLineIsBlank = False

End Function

Private Function IsAutomationDisconnectError(ByVal errorNumber As Long) As Boolean

    Select Case errorNumber

        Case -2147418105, -2147417848, -2147417846, 440, 91
            IsAutomationDisconnectError = True

        Case Else
            IsAutomationDisconnectError = False

    End Select

End Function


'=============================================================
' DOCUMENT NUMBERING
'=============================================================

Private Function GetNextInvoiceNumber() As String

    On Error GoTo Err_Handler

    Dim db As DAO.Database
    Dim rs As DAO.Recordset
    Dim sqlLookup As String

    Dim prefixValue As String
    Dim nextNumberValue As Long
    Dim numberLengthValue As Long

    Set db = CurrentDb

    sqlLookup = _
        "SELECT * FROM T_DocumentNumbers " & _
        "WHERE DocumentType = 'Invoice';"

    Set rs = db.OpenRecordset(sqlLookup, dbOpenDynaset)

    If rs.EOF Then

        prefixValue = "INV"
        nextNumberValue = 157
        numberLengthValue = 4

        rs.addnew
        rs!documentType = "Invoice"
        rs!prefix = prefixValue
        rs!nextNumber = nextNumberValue + 1
        rs!NumberLength = numberLengthValue
        rs!LastUpdatedOn = Now()
        rs.Update

    Else

        prefixValue = Nz(rs!prefix, "INV")
        nextNumberValue = Nz(rs!nextNumber, 157)
        numberLengthValue = Nz(rs!NumberLength, 4)

        rs.Edit
        rs!nextNumber = nextNumberValue + 1
        rs!LastUpdatedOn = Now()
        rs.Update

    End If

    GetNextInvoiceNumber = _
        prefixValue & Format(nextNumberValue, String(numberLengthValue, "0"))

Exit_Handler:

    On Error Resume Next

    If Not rs Is Nothing Then
        rs.close
    End If

    Set rs = Nothing
    Set db = Nothing

    On Error GoTo 0

    Exit Function

Err_Handler:
    GetNextInvoiceNumber = "INV" & Format(Now(), "yyyymmddhhnnss")
    Resume Exit_Handler

End Function

Private Function CleanFileName(ByVal fileNameText As String) As String

    Dim badChars As Variant
    Dim i As Long

    badChars = Array("\", "/", ":", "*", "?", """", "<", ">", "|")

    CleanFileName = Trim$(Nz(fileNameText, vbNullString))

    For i = LBound(badChars) To UBound(badChars)
        CleanFileName = Replace(CleanFileName, badChars(i), "_")
    Next i

    If Len(CleanFileName) = 0 Then
        CleanFileName = "Invoice"
    End If

End Function
