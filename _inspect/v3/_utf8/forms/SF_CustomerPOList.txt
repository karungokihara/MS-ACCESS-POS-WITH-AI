Public Function OpenSelectedRecord() As Boolean

    On Error GoTo Err_Handler

    If Nz(Me!CustomerPOID, 0) <= 0 Then Exit Function

    DoCmd.OpenForm "F_CustomerPO", acNormal, , "CustomerPOID = " & CLng(Me!CustomerPOID)

    OpenSelectedRecord = True

Exit_Handler:
    Exit Function

Err_Handler:
    MsgBox "The selected customer PO could not be opened." & vbCrLf & vbCrLf & _
           "Error Number: " & Err.Number & vbCrLf & _
           "Details: " & Err.Description, vbExclamation, "Open Customer PO"
    Resume Exit_Handler

End Function
